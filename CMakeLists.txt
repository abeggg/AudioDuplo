cmake_minimum_required(VERSION 3.21)
project(AudioDuplo VERSION 1.1.4 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "AudioDuplo requires Windows and the Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(audio_core STATIC src/AudioEngine.cpp)
target_include_directories(audio_core PUBLIC src)
target_compile_definitions(audio_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(audio_core PUBLIC ole32 uuid ksuser avrt)
if(MSVC)
  target_compile_options(audio_core PUBLIC /W4 /permissive- /EHsc /utf-8)
endif()
add_library(audio_ui STATIC src/UiRenderer.cpp src/ControlSurface.cpp)
target_include_directories(audio_ui PUBLIC src)
target_compile_definitions(audio_ui PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(audio_ui PUBLIC gdiplus user32 gdi32)
if(MSVC)
  target_compile_options(audio_ui PUBLIC /W4 /permissive- /EHsc /utf-8)
endif()
add_executable(AudioDuplo WIN32 src/main.cpp resources/app.rc resources/fonts.rc)
target_link_libraries(AudioDuplo PRIVATE audio_core audio_ui user32 gdi32 comctl32 shell32 dwmapi uxtheme)
target_include_directories(AudioDuplo PRIVATE resources)
set_target_properties(AudioDuplo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_link_options(AudioDuplo PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(audio_tests tests/audio_tests.cpp)
  target_link_libraries(audio_tests PRIVATE audio_core)
  add_test(NAME audio_core COMMAND audio_tests)
  add_executable(audio_probe tests/audio_probe.cpp)
  target_link_libraries(audio_probe PRIVATE audio_core)
  add_executable(ui_preview tests/ui_preview.cpp resources/fonts.rc)
  target_include_directories(ui_preview PRIVATE resources)
  target_link_libraries(ui_preview PRIVATE audio_ui)
  add_test(NAME ui_layout COMMAND ui_preview --test)
  add_executable(ui_native_tests tests/ui_native_tests.cpp)
  target_link_libraries(ui_native_tests PRIVATE audio_ui comctl32 gdi32)
  add_test(NAME ui_native COMMAND ui_native_tests)
endif()
install(TARGETS AudioDuplo RUNTIME DESTINATION .)
install(FILES README.md LICENSE NOTICE.md CHANGELOG.md CONTRIBUTING.md SECURITY.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES resources/fonts/Nunito-OFL.txt DESTINATION licenses RENAME Nunito-OFL.txt)
